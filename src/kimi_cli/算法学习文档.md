#### 统一差异格式化 (Unified Diff)
```python
def format_unified_diff(old_text: str, new_text: str, path: str = "") -> str:
    old_lines = old_text.splitlines(keepends=True)
    new_lines = new_text.splitlines(keepends=True)

    # 确保行尾有换行符
    if old_lines and not old_lines[-1].endswith("\n"):
        old_lines[-1] += "\n"

    diff = list(difflib.unified_diff(
        old_lines, new_lines,
        fromfile=f"a/{path}",
        tofile=f"b/{path}",
        lineterm="\n",
    ))
```

**算法要点**:
- 使用 Python 标准库 `difflib.unified_diff`
- 保持原始行尾符 (`keepends=True`)
- 生成标准 Unix diff 格式

#### 分块差异分析 (SequenceMatcher)
```python
def build_diff_blocks(path: str, old_text: str, new_text: str) -> list[DiffDisplayBlock]:
    old_lines = old_text.splitlines()
    new_lines = new_text.splitlines()
    matcher = SequenceMatcher(None, old_lines, new_lines, autojunk=False)

    for group in matcher.get_grouped_opcodes(n=N_CONTEXT_LINES):
        i1, i2 = group[0][1], group[-1][2]  # 旧文本范围
        j1, j2 = group[0][3], group[-1][4]  # 新文本范围
        blocks.append(DiffDisplayBlock(
            path=path,
            old_text="\n".join(old_lines[i1:i2]),
            new_text="\n".join(new_lines[j1:j2]),
        ))
```

**算法要点**:
- `SequenceMatcher`: 基于 Ratcliff/Obershelp 算法的序列比较
- `autojunk=False`: 禁用启发式垃圾检测，提高准确性
- `get_grouped_opcodes()`: 将操作码分组，提供上下文窗口
- 操作码类型: `'equal'`, `'delete'`, `'insert'`, `'replace'`

### 2.3 学习重点
1. 理解 `SequenceMatcher` 的工作原理
2. 掌握统一差异格式的结构

```python
class FlowNode:
    id: str
    label: str | list[ContentPart]
    kind: FlowNodeKind  # "begin" | "end" | "task" | "decision"

@dataclass(frozen=True, slots=True)

    dst: str   # 目标节点ID
    label: str | None  # 边标签（用于决策分支）

@dataclass(slots=True)
class Flow:
    nodes: dict[str, FlowNode]
    outgoing: dict[str, list[FlowEdge]]
    begin_id: str
    end_id: str
```

### 4.3 流程图验证算法 (BFS)

```python
def validate_flow(nodes: dict[str, FlowNode], outgoing: dict[str, list[FlowEdge]])
    -> tuple[str, str]:

    # 1. 验证 BEGIN/END 节点唯一性
    begin_ids = [node.id for node in nodes.values() if node.kind == "begin"]
    end_ids = [node.id for node in nodes.values() if node.kind == "end"]

    if len(begin_ids) != 1:
        raise FlowValidationError(f"Expected exactly one BEGIN node")

    # 2. BFS 可达性分析
    reachable: set[str] = set()
    queue: list[str] = [begin_id]
    while queue:
        node_id = queue.pop()
        if node_id in reachable:
            continue
        reachable.add(node_id)
        for edge in outgoing.get(node_id, []):
            if edge.dst not in reachable:
                queue.append(edge.dst)

    # 3. 验证 END 节点可达
    if end_id not in reachable:
        raise FlowValidationError("END node is not reachable from BEGIN")

    # 4. 验证决策节点边标签唯一性
    for node in nodes.values():
        edges = outgoing.get(node.id, [])
        if len(edges) > 1:
            labels = [edge.label for edge in edges if edge.label]
            if len(set(labels)) != len(labels):
                raise FlowValidationError(f'Node has duplicate edge labels')
```

**算法要点**:
- BFS 遍历验证图连通性
- 检查决策节点边标签唯一性（防止歧义）
- 确保流程有明确的开始和结束

### 4.4 Mermaid 语法解析 (递归下降)

```python
def parse_mermaid_flowchart(text: str) -> Flow:
    nodes: dict[str, _NodeDef] = {}
    outgoing: dict[str, list[FlowEdge]] = {}

    for line_no, raw_line in enumerate(text.splitlines(), start=1):
        line = _strip_comment(raw_line).strip()

        # 尝试解析边: A --> B
        edge = _try_parse_edge_line(line, line_no)
        if edge is not None:
            src_spec, label, dst_spec = edge
            src_node = _add_node(nodes, src_spec, line_no)
            dst_node = _add_node(nodes, dst_spec, line_no)
            flow_edge = FlowEdge(src=src_node.id, dst=dst_node.id, label=label)
            outgoing.setdefault(flow_edge.src, []).append(flow_edge)
            continue

        # 尝试解析节点: A[Label]
        node_spec = _try_parse_node_line(line, line_no)
        if node_spec is not None:
            _add_node(nodes, node_spec, line_no)

    # 推断决策节点（出边数 > 1 的任务节点）
    flow_nodes = _infer_decision_nodes(flow_nodes, outgoing)
```

```python
def _parse_node_token(line: str, idx: int, line_no: int) -> tuple[_NodeSpec, int]:
    # 匹配节点ID: [A-Za-z0-9_][A-Za-z0-9_-]*
    match = _NODE_ID_RE.match(line, idx)
    node_id = match.group(0)
    idx = match.end()

    # 检查是否有标签: A[label] 或 A(label) 或 A{label}
    if idx >= len(line) or line[idx] not in _SHAPES:
        return _NodeSpec(node_id=node_id, label=None), idx

    close_char = _SHAPES[line[idx]]  # '[' -> ']'
    idx += 1
    label, idx = _parse_label(line, idx, close_char, line_no)
    return _NodeSpec(node_id=node_id, label=label), idx
```

```python
class SimpleCompaction:
    def __init__(self, max_preserved_messages: int = 2):
        self.max_preserved_messages = max_preserved_messages

    async def compact(self, messages: Sequence[Message], llm: LLM) -> Sequence[Message]:
        compact_message, to_preserve = self.prepare(messages)
        if compact_message is None:
            return to_preserve

        # 调用 LLM 进行摘要
        result = await kosong.step(
            chat_provider=llm.chat_provider,
            system_prompt="You are a helpful assistant that compacts conversation context.",
            toolset=EmptyToolset(),
            history=[compact_message],
        )

        # 构建压缩后的消息序列
        compacted_messages = [Message(role="user", content=[...])]
        compacted_messages.extend(to_preserve)
        return compacted_messages
```

### 5.3 消息分割策略

```python
def prepare(self, messages: Sequence[Message]) -> PrepareResult:
    # 从后向前遍历，保留最近 N 条用户/助手消息
    history = list(messages)
    preserve_start_index = len(history)
    n_preserved = 0

    for index in range(len(history) - 1, -1, -1):
        if history[index].role in {"user", "assistant"}:
            n_preserved += 1
            if n_preserved == self.max_preserved_messages:
                preserve_start_index = index
                break

    to_compact = history[:preserve_start_index]   # 需要压缩的历史
    to_preserve = history[preserve_start_index:]  # 保留的最近消息

    # 构建压缩输入消息
    compact_message = Message(role="user", content=[])
    for i, msg in enumerate(to_compact):
        compact_message.content.append(
            TextPart(text=f"## Message {i + 1}\nRole: {msg.role}\nContent:\n")
        )
        compact_message.content.extend(msg.content)
```

**算法要点**:
- 保留策略: 始终保留最近 N 条对话消息
- 压缩策略: 使用 LLM 对历史消息进行摘要
- 触发条件: 上下文长度接近模型限制时触发

```python
async def _step(self) -> StepOutcome | None:
    # 带重试机制的 LLM 调用
    @tenacity.retry(
        retry=retry_if_exception(self._is_retryable_error),
        wait=wait_exponential_jitter(initial=0.3, max=5, jitter=0.5),
        stop=stop_after_attempt(self._loop_control.max_retries_per_step),
    )
    async def _kosong_step_with_retry() -> StepResult:
        return await kosong.step(
            chat_provider,
            self._agent.system_prompt,
            self._agent.toolset,
            self._context.history,
            on_message_part=wire_send,
            on_tool_result=wire_send,
        )

    result = await _kosong_step_with_retry()

    # 等待工具执行结果
    results = await result.tool_results()

    # 更新上下文
    await self._grow_context(result, results)

    # 判断是否需要继续

        return None  # 继续下一步
    return StepOutcome(stop_reason="no_tool_calls", ...)
```

### 6.4 流程图执行算法

```python
async def run(self, soul: KimiSoul, args: str) -> None:
    current_id = self._flow.begin_id
    moves = 0

    while True:
        node = self._flow.nodes[current_id]
        edges = self._flow.outgoing.get(current_id, [])

        if node.kind == "end":
            return  # 流程结束

        if node.kind == "begin":
            current_id = edges[0].dst
            continue

        # 执行节点并获取下一步
        next_id, steps_used = await self._execute_flow_node(soul, node, edges)
        if next_id is None:
            return
        moves += 1
        current_id = next_id
```

```python
def shorten_middle(text: str, width: int, remove_newline: bool = True) -> str:
    """Shorten the text by inserting ellipsis in the middle."""
    if len(text) <= width:
        return text
    if remove_newline:
        text = _NEWLINE_RE.sub(" ", text)
    return text[: width // 2] + "..." + text[-width // 2 :]
```

## 8. 异步队列算法

### 8.1 文件位置
`utils/aioqueue.py`

### 8.2 支持关闭信号的队列

```python
class Queue[T](asyncio.Queue[T | _Shutdown]):
    """Asyncio Queue with shutdown support for Python < 3.13."""

    def shutdown(self, immediate: bool = False) -> None:
        if self._shutdown:
            return
        self._shutdown = True
        if immediate:
            self._queue.clear()

        # 向等待的消费者发送关闭信号
        getters = list(getattr(self, "_getters", []))
        count = max(1, len(getters))
        self._enqueue_shutdown(count)

    async def get(self) -> T:
        if self._shutdown and self.empty():
            raise QueueShutDown
        item = await super().get()
        if isinstance(item, _Shutdown):
            raise QueueShutDown
        return item
```

**算法要点**:
- 使用哨兵对象 (`_Shutdown`) 传递关闭信号
- 兼容 Python 3.13+ 和旧版本
- 优雅地通知所有等待中的消费者

```python
class WireMessageEnvelope(BaseModel):
    type: str
    payload: dict[str, JsonType]

    @classmethod
    def from_wire_message(cls, msg: WireMessage) -> WireMessageEnvelope:
        # 根据消息类型查找对应类
        for name, typ in _NAME_TO_WIRE_MESSAGE_TYPE.items():
            if issubclass(type(msg), typ):

                break
```
